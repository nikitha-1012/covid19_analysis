Demonstration Aggregation Pipelines

Calculations executed inside the MongoDB Atlas distributed query engine:

1. Worldwide KPI Aggregation Pipeline 
[
  {
    "$group": {
      "_id": null,
      "total_confirmed": { "$sum": "$confirmed" },
      "total_deaths": { "$sum": "$deaths" },
      "total_active": { "$sum": "$active" },
      "total_countries": { "$sum": 1 }
    }
  },
  {
    "$project": {
      "_id": 0,
      "total_confirmed": 1,
      "total_deaths": 1,
      "mortality_rate": {
        "$round": [
          { "$multiply": [{ "$divide": ["$total_deaths", "$total_confirmed"] }, 100] },
          2
        ]
      }
    }
  }
]
2. Continental Rollup Pipeline 
[
  { "$match": { "continent": { "$nin": ["Unknown", null] } } },
  {
    "$group": {
      "_id": "$continent",
      "total_confirmed": { "$sum": "$confirmed" },
      "total_deaths": { "$sum": "$deaths" },
      "country_count": { "$sum": 1 }
    }
  },
  {
    "$project": {
      "_id": 0,
      "continent": "$_id",
      "total_confirmed": 1,
      "total_deaths": 1,
      "country_count": 1,
      "mortality_rate": {
        "$round": [
          { "$multiply": [{ "$divide": ["$total_deaths", "$total_confirmed"] }, 100] },
          2
        ]
      }
    }
  },
  { "$sort": { "total_confirmed": -1 } }
]
